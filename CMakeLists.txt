cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(thermal_uvc C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(PICO_SDK_VERSION_STRING VERSION_LESS "2.3.1")
 message(FATAL_ERROR "Pico SDK 2.3.1 or newer required")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/lib/mlx90640-library/functions/MLX90640_API.c")
 message(FATAL_ERROR "Run git submodule update --init --recursive")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(TINYUSB_UVC_DESC "${PICO_TINYUSB_PATH}/examples/device/video_capture/src/usb_descriptors.c")
if(NOT EXISTS "${TINYUSB_UVC_DESC}")
 message(FATAL_ERROR "TinyUSB video_capture descriptor not found at ${TINYUSB_UVC_DESC}")
endif()
set(GENERATED_DESC "${CMAKE_CURRENT_BINARY_DIR}/usb_descriptors.c")
add_custom_command(OUTPUT "${GENERATED_DESC}" COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/generate_usb_descriptor.py" "${TINYUSB_UVC_DESC}" "${GENERATED_DESC}" DEPENDS "${TINYUSB_UVC_DESC}" "${CMAKE_CURRENT_LIST_DIR}/generate_usb_descriptor.py")
add_executable(thermal_uvc src/main.cpp src/thermal_pipeline.cpp src/status_screens.cpp src/mlx90640_pico_i2c.cpp  ${PICO_TINYUSB_PATH}/examples/device/video_capture/src/usb_descriptors.c lib/mlx90640-library/functions/MLX90640_API.c)
target_include_directories(thermal_uvc PRIVATE src lib/mlx90640-library/headers)
target_compile_definitions(thermal_uvc PRIVATE CFG_TUSB_CONFIG_FILE="tusb_config.h" CFG_EXAMPLE_VIDEO_DISABLE_MJPEG=1)
target_compile_options(thermal_uvc PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(thermal_uvc PRIVATE -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/thermal_uvc.map)
target_link_libraries(thermal_uvc PRIVATE pico_stdlib pico_multicore hardware_i2c tinyusb_device tinyusb_board)
pico_enable_stdio_usb(thermal_uvc 0)
pico_enable_stdio_uart(thermal_uvc 0)
pico_add_extra_outputs(thermal_uvc)
